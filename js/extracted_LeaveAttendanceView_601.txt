// ----------------------------------------------------
// Leave & Attendance Tracking View Component
// ----------------------------------------------------
function LeaveAttendanceView({ leaveRequests, attendanceLogs, members, currentUser, onOpenLeaveModal, onOpenAttendanceModal, onApproveLeave, onDeleteLeave, onDeleteAttendance }) {
  const [activeTab, setActiveTab] = useState('schedule'); // 'schedule' | 'requests' | 'attendance'

  // Summary by staff member
  const staffSummary = useMemo(() => {
    return members.map(m => {
      const mLeaves = leaveRequests.filter(l => l.employeeName === m.name && l.status === '✅ อนุมัติแล้ว');
      const sickDays = mLeaves.filter(l => l.leaveType.includes('ป่วย')).reduce((sum, l) => sum + (l.totalDays || 1), 0);
      const personalDays = mLeaves.filter(l => l.leaveType.includes('กิจ')).reduce((sum, l) => sum + (l.totalDays || 1), 0);
      const vacationDays = mLeaves.filter(l => l.leaveType.includes('พักร้อน')).reduce((sum, l) => sum + (l.totalDays || 1), 0);

      const mAtt = attendanceLogs.filter(a => a.employeeName === m.name);
      const lateMins = mAtt.filter(a => a.type.includes('สาย')).reduce((sum, a) => sum + (a.lateMinutes || 0), 0);
      const absentTimes = mAtt.filter(a => a.type.includes('ขาด')).length;
      const totalFine = mAtt.reduce((sum, a) => sum + (a.fineAmount || 0), 0);

      return {
        ...m,
        sickDays,
        personalDays,
        vacationDays,
        lateMins,
        absentTimes,
        totalFine
      };
    });
  }, [members, leaveRequests, attendanceLogs]);

  return (
    <div className="space-y-6">
      {/* Header Banner */}
      <div className="glass-panel p-5 rounded-2xl flex flex-col md:flex-row md:items-center justify-between gap-4 border border-slate-800">
        <div className="flex items-center gap-3">
          <div className="w-12 h-12 rounded-2xl bg-rose-600/20 border border-rose-500/30 flex items-center justify-center text-2xl shadow-inner">
            📅
          </div>
          <div>
            <h2 className="text-lg font-bold text-white flex items-center gap-2">
              <span>ระบบตารางวันลา & ขาด ลา มาสาย (Leave & Attendance Schedule)</span>
              <span className="text-xs px-2.5 py-0.5 rounded-full bg-rose-500/20 text-rose-300 border border-rose-500/30 font-mono">
                บุคลากร {members.length} ท่าน
              </span>
            </h2>
            <p className="text-xs text-slate-400">
              ลงวันลาป่วย/ลากิจ/ลาพักร้อน ตรวจสอบคิวลาป้องกันงานชนกัน และลงบันทึก ขาด-มาสาย พร้อมสรุปค่าปรับ
            </p>
          </div>
        </div>

        <div className="flex flex-wrap items-center gap-2">
          <button
            type="button"
            onClick={onOpenLeaveModal}
            className="px-4 py-2 bg-emerald-600 hover:bg-emerald-500 text-white font-bold rounded-xl text-xs shadow-md shadow-emerald-600/30 flex items-center gap-1.5"
          >
            <span>+ ลงวันลาใหม่</span>
          </button>
          <button
            type="button"
            onClick={onOpenAttendanceModal}
            className="px-4 py-2 bg-rose-600 hover:bg-rose-500 text-white font-bold rounded-xl text-xs shadow-md shadow-rose-600/30 flex items-center gap-1.5"
          >
            <span>+ บันทึก ขาด / มาสาย</span>
          </button>
        </div>
      </div>

      {/* Staff Quota & Attendance Balance Cards */}
      <div className="grid grid-cols-1 md:grid-cols-2 lg:grid-cols-3 gap-4">
        {staffSummary.map(staff => (
          <div key={staff.id} className="glass-card p-4 rounded-2xl border border-slate-800 space-y-3 hover:border-indigo-500/40 transition-colors">
            <div className="flex items-center justify-between">
              <div className="flex items-center gap-2.5">
                <span className="text-2xl">{staff.avatar || '👤'}</span>
                <div>
                  <h3 className="font-extrabold text-white text-sm">{staff.name}</h3>
                  <span className="text-[11px] text-amber-300 font-medium">{staff.role}</span>
                </div>
              </div>
              {staff.totalFine > 0 && (
                <span className="text-[11px] font-mono font-bold text-rose-400 bg-rose-950/60 px-2 py-0.5 rounded-lg border border-rose-800/60">
                  หัก {staff.totalFine.toLocaleString()} บ.
                </span>
              )}
            </div>

            {/* Leave Usage Gauges */}
            <div className="grid grid-cols-3 gap-2 text-center text-[11px]">
              <div className="bg-slate-950 p-2 rounded-xl border border-slate-800">
                <div className="text-slate-400">🤒 ป่วย</div>
                <div className="font-bold text-indigo-300 mt-0.5">{staff.sickDays}/30 <span className="text-[9px] text-slate-500">วัน</span></div>
              </div>
              <div className="bg-slate-950 p-2 rounded-xl border border-slate-800">
                <div className="text-slate-400">🌴 กิจ</div>
                <div className="font-bold text-amber-300 mt-0.5">{staff.personalDays}/6 <span className="text-[9px] text-slate-500">วัน</span></div>
              </div>
              <div className="bg-slate-950 p-2 rounded-xl border border-slate-800">
                <div className="text-slate-400">🏖️ พักร้อน</div>
                <div className="font-bold text-emerald-300 mt-0.5">{staff.vacationDays}/6 <span className="text-[9px] text-slate-500">วัน</span></div>
              </div>
            </div>

            {/* Attendance Record */}
            <div className="flex items-center justify-between text-xs pt-1 border-t border-slate-800/80 text-slate-400">
              <span>⏰ สายสะสม: <strong className="text-amber-300 font-mono">{staff.lateMins} นาที</strong></span>
              <span>🚫 ขาดงาน: <strong className="text-rose-400 font-mono">{staff.absentTimes} ครั้ง</strong></span>
            </div>
          </div>
        ))}
      </div>

      {/* Sub Tabs Switcher */}
      <div className="flex bg-slate-900 p-1.5 rounded-2xl border border-slate-800 text-xs w-fit">
        <button
          onClick={() => setActiveTab('schedule')}
          className={`px-4 py-2 rounded-xl font-bold transition-all ${
            activeTab === 'schedule' ? 'bg-indigo-600 text-white shadow-md' : 'text-slate-400 hover:text-white'
          }`}
        >
          📅 ตารางวันลาของทีม ({leaveRequests.filter(l => l.status === '✅ อนุมัติแล้ว').length})
        </button>
        <button
          onClick={() => setActiveTab('requests')}
          className={`px-4 py-2 rounded-xl font-bold transition-all ${
            activeTab === 'requests' ? 'bg-indigo-600 text-white shadow-md' : 'text-slate-400 hover:text-white'
          }`}
        >
          📜 คำขอลาทั้งหมด ({leaveRequests.length})
        </button>
        <button
          onClick={() => setActiveTab('attendance')}
          className={`px-4 py-2 rounded-xl font-bold transition-all ${
            activeTab === 'attendance' ? 'bg-indigo-600 text-white shadow-md' : 'text-slate-400 hover:text-white'
          }`}
        >
          ⏰ บันทึก ขาด / มาสาย ({attendanceLogs.length})
        </button>
      </div>

      {/* SUB TAB 1: Leave Schedule Table */}
      {activeTab === 'schedule' && (
        <div className="glass-panel p-5 rounded-2xl border border-slate-800 space-y-4">
          <div className="flex items-center justify-between border-b border-slate-800 pb-3">
            <h3 className="font-extrabold text-white text-sm flex items-center gap-2">
              <span>📅 ปฏิทิน/ตารางแสดงวันลาของบุคลากร (ป้องกันคิวงานชนกัน)</span>
            </h3>
          </div>

          <div className="overflow-x-auto">
            <table className="w-full text-left text-xs text-slate-300">
              <thead className="bg-slate-950 uppercase text-[10px] text-slate-400 border-b border-slate-800">
                <tr>
                  <th className="p-3">ผู้ขอลา</th>
                  <th className="p-3">ประเภทการลา</th>
                  <th className="p-3">ช่วงวันที่ลา</th>
                  <th className="p-3 text-center">จำนวนวัน</th>
                  <th className="p-3">เหตุผลการลา</th>
                  <th className="p-3">ผู้อนุมัติ</th>
                </tr>
              </thead>
              <tbody className="divide-y divide-slate-800">
                {leaveRequests.filter(l => l.status === '✅ อนุมัติแล้ว').map(req => (
                  <tr key={req.id} className="hover:bg-slate-900/40">
                    <td className="p-3 font-bold text-amber-300">{req.employeeName}</td>
                    <td className="p-3 font-bold text-indigo-300">{req.leaveType}</td>
                    <td className="p-3 font-mono text-slate-200">{req.startDate} ถึง {req.endDate}</td>
                    <td className="p-3 text-center font-bold text-emerald-400 font-mono">{req.totalDays} วัน</td>
                    <td className="p-3 text-slate-300">{req.reason}</td>
                    <td className="p-3 text-slate-400">{req.approvedBy || '-'}</td>
                  </tr>
                ))}
              </tbody>
            </table>
          </div>
        </div>
      )}

      {/* SUB TAB 2: Leave Requests List */}
      {activeTab === 'requests' && (
        <div className="glass-panel p-5 rounded-2xl border border-slate-800 space-y-4">
          <div className="flex items-center justify-between border-b border-slate-800 pb-3">
            <h3 className="font-extrabold text-white text-sm flex items-center gap-2">
              <span>📜 รายการคำขอลาและสถานะอนุมัติ</span>
            </h3>
          </div>

          <div className="overflow-x-auto">
            <table className="w-full text-left text-xs text-slate-300">
              <thead className="bg-slate-950 uppercase text-[10px] text-slate-400 border-b border-slate-800">
                <tr>
                  <th className="p-3">ผู้ยื่นคำขอ</th>
                  <th className="p-3">ประเภท</th>
                  <th className="p-3">วันที่เริ่ม - สิ้นสุด</th>
                  <th className="p-3 text-center">จำนวน</th>
                  <th className="p-3">เหตุผล</th>
                  <th className="p-3">สถานะ</th>
                  <th className="p-3 text-right">จัดการ</th>
                </tr>
              </thead>
              <tbody className="divide-y divide-slate-800">
                {leaveRequests.map(req => (
                  <tr key={req.id} className="hover:bg-slate-900/40">
                    <td className="p-3 font-bold text-amber-300">{req.employeeName}</td>
                    <td className="p-3 font-bold text-indigo-300">{req.leaveType}</td>
                    <td className="p-3 font-mono">{req.startDate} ถึง {req.endDate}</td>
                    <td className="p-3 text-center font-bold text-emerald-400">{req.totalDays} วัน</td>
                    <td className="p-3 text-slate-300">{req.reason}</td>
                    <td className="p-3">
                      <span className={`px-2.5 py-0.5 rounded-full text-[11px] font-bold border ${
                        req.status === '✅ อนุมัติแล้ว' ? 'bg-emerald-500/20 text-emerald-300 border-emerald-500/30' :
                        req.status === '❌ ไม่อนุมัติ' ? 'bg-rose-500/20 text-rose-300 border-rose-500/30' :
                        'bg-amber-500/20 text-amber-300 border-amber-500/30 animate-pulse'
                      }`}>
                        {req.status}
                      </span>
                    </td>
                    <td className="p-3 text-right space-x-2">
                      {req.status === '⏳ รออนุมัติ' && (currentUser?.role === 'owner' || currentUser?.role === 'admin_lead') && (
                        <>
                          <button
                            onClick={() => onApproveLeave(req.id, '✅ อนุมัติแล้ว')}
                            className="px-2.5 py-1 bg-emerald-600 hover:bg-emerald-500 text-white font-bold rounded-lg text-[11px]"
                          >
                            อนุมัติ
                          </button>
                          <button
                            onClick={() => onApproveLeave(req.id, '❌ ไม่อนุมัติ')}
                            className="px-2.5 py-1 bg-rose-600 hover:bg-rose-500 text-white font-bold rounded-lg text-[11px]"
                          >
                            ปฏิเสธ
                          </button>
                        </>
                      )}
                      <button
                        onClick={() => onDeleteLeave(req.id)}
                        className="px-2 py-1 bg-slate-800 hover:bg-slate-700 text-slate-400 hover:text-white rounded-lg text-[11px]"
                      >
                        🗑️
                      </button>
                    </td>
                  </tr>
                ))}
              </tbody>
            </table>
          </div>
        </div>
      )}

      {/* SUB TAB 3: Attendance Logs */}
      {activeTab === 'attendance' && (
        <div className="glass-panel p-5 rounded-2xl border border-slate-800 space-y-4">
          <div className="flex items-center justify-between border-b border-slate-800 pb-3">
            <h3 className="font-extrabold text-white text-sm flex items-center gap-2">
              <span>⏰ ประวัติ ขาด / มาสาย และค่าปรับหักเงิน</span>
            </h3>
          </div>

          <div className="overflow-x-auto">
            <table className="w-full text-left text-xs text-slate-300">
              <thead className="bg-slate-950 uppercase text-[10px] text-slate-400 border-b border-slate-800">
                <tr>
                  <th className="p-3">วันที่</th>
                  <th className="p-3">พนักงาน</th>
                  <th className="p-3">ประเภท</th>
                  <th className="p-3 text-center">นาทีที่สาย</th>
                  <th className="p-3 text-right">ยอดหักเงิน</th>
                  <th className="p-3">หมายเหตุ</th>
                  <th className="p-3 text-right">ลบ</th>
                </tr>
              </thead>
              <tbody className="divide-y divide-slate-800">
                {attendanceLogs.map(att => (
                  <tr key={att.id} className="hover:bg-slate-900/40">
                    <td className="p-3 font-mono text-slate-400">{att.date}</td>
                    <td className="p-3 font-bold text-amber-300">{att.employeeName}</td>
                    <td className="p-3 font-bold text-rose-400">{att.type}</td>
                    <td className="p-3 text-center font-mono">{att.lateMinutes > 0 ? `${att.lateMinutes} นาที` : '-'}</td>
                    <td className="p-3 text-right font-mono font-bold text-rose-400">
                      {att.fineAmount > 0 ? `${att.fineAmount.toLocaleString()} บาท` : '-'}
                    </td>
                    <td className="p-3 text-slate-300">{att.note}</td>
                    <td className="p-3 text-right">
                      <button
                        onClick={() => onDeleteAttendance(att.id)}
                        className="px-2 py-1 bg-slate-800 hover:bg-slate-700 text-rose-400 rounded-lg text-[11px]"
                      >
                        🗑️
                      </button>
                    </td>
                  </tr>
                ))}
              </tbody>
            </table>
          </div>
        </div>
      )}
    </div>
  );
}

// ----------------------------------------------------
// Leave Request Modal Component
// ----------------------------------------------------
function LeaveRequestModal({ members, currentUser, onSave, onClose }) {
  const [employeeName, setEmployeeName] = useState(currentUser?.fullName || members[0]?.name || '');
  const [leaveType, setLeaveType] = useState('🤒 ลาป่วย');
  const [startDate, setStartDate] = useState(new Date().toISOString().substring(0, 10));
  const [endDate, setEndDate] = useState(new Date().toISOString().substring(0, 10));
  const [totalDays, setTotalDays] = useState(1);
  const [reason, setReason] = useState('');

  const handleSubmit = (e) => {
    e.preventDefault();
    if (!reason.trim()) {
      alert('กรุณากรอกเหตุผลการลา');
      return;
    }
    onSave({
      id: 'leave-' + Date.now(),
      employeeName,
      leaveType,
      startDate,
      endDate,
      totalDays: Number(totalDays) || 1,
      reason,
      status: (currentUser?.role === 'owner' || currentUser?.role === 'admin_lead') ? '✅ อนุมัติแล้ว' : '⏳ รออนุมัติ',
      approvedBy: (currentUser?.role === 'owner' || currentUser?.role === 'admin_lead') ? currentUser?.fullName : '',
      createdAt: new Date().toISOString().replace('T', ' ').substring(0, 16)
    });
  };

  return (
    <div className="fixed inset-0 z-50 bg-slate-950/80 backdrop-blur-sm flex items-center justify-center p-4">
      <div className="bg-slate-900 border border-slate-800 w-full max-w-lg rounded-2xl p-6 space-y-4 shadow-2xl animate-modal text-xs">
        <div className="flex items-center justify-between border-b border-slate-800 pb-3">
          <h3 className="font-extrabold text-white text-base flex items-center gap-2">
            <span>📝 ยื่นคำขอลา / ลงตารางวันลาใหม่</span>
          </h3>
          <button onClick={onClose} className="text-slate-400 hover:text-white text-base">✕</button>
        </div>

        <form onSubmit={handleSubmit} className="space-y-4">
          <div className="space-y-1">
            <label className="text-slate-300 font-bold">ชื่อพนักงาน / บุคลากร:</label>
            <select
              value={employeeName}
              onChange={(e) => setEmployeeName(e.target.value)}
              className="w-full bg-slate-950 border border-slate-800 rounded-xl p-2.5 text-slate-100 outline-none"
            >
              {members.map(m => (
                <option key={m.id} value={m.name}>{m.name} ({m.role})</option>
              ))}
            </select>
          </div>

          <div className="grid grid-cols-2 gap-3">
            <div className="space-y-1">
              <label className="text-slate-300 font-bold">ประเภทการลา:</label>
              <select
                value={leaveType}
                onChange={(e) => setLeaveType(e.target.value)}
                className="w-full bg-slate-950 border border-slate-800 rounded-xl p-2.5 text-slate-100 outline-none font-semibold"
              >
                <option value="🤒 ลาป่วย">🤒 ลาป่วย (Sick Leave)</option>
                <option value="🌴 ลากิจ">🌴 ลากิจ (Personal Leave)</option>
                <option value="🏖️ ลาพักร้อน">🏖️ ลาพักร้อน (Vacation Leave)</option>
                <option value="👶 ลาคลอด / ลาบวช">👶 ลาคลอด / ลาบวช / อื่นๆ</option>
              </select>
            </div>
            <div className="space-y-1">
              <label className="text-slate-300 font-bold">จำนวนวันลา (วัน):</label>
              <input
                type="number"
                min="0.5"
                step="0.5"
                value={totalDays}
                onChange={(e) => setTotalDays(e.target.value)}
                className="w-full bg-slate-950 border border-slate-800 rounded-xl p-2.5 text-slate-100 outline-none font-mono"
              />
            </div>
          </div>

          <div className="grid grid-cols-2 gap-3">
            <div className="space-y-1">
              <label className="text-slate-300 font-bold">วันที่เริ่มลา:</label>
              <input
                type="date"
                value={startDate}
                onChange={(e) => setStartDate(e.target.value)}
                className="w-full bg-slate-950 border border-slate-800 rounded-xl p-2.5 text-slate-100 outline-none font-mono"
              />
            </div>
            <div className="space-y-1">
              <label className="text-slate-300 font-bold">วันที่สิ้นสุด:</label>
              <input
                type="date"
                value={endDate}
                onChange={(e) => setEndDate(e.target.value)}
                className="w-full bg-slate-950 border border-slate-800 rounded-xl p-2.5 text-slate-100 outline-none font-mono"
              />
            </div>
          </div>

          <div className="space-y-1">
            <label className="text-slate-300 font-bold">เหตุผลการลา / รายละเอียด:</label>
            <textarea
              rows={3}
              placeholder="ระบุเหตุผลการลา เช่น เป็นไข้หวัดมีใบรับรองแพทย์ หรือ ติดธุระ..."
              value={reason}
              onChange={(e) => setReason(e.target.value)}
              className="w-full bg-slate-950 border border-slate-800 rounded-xl p-2.5 text-slate-100 outline-none"
            />
          </div>

          <div className="flex items-center justify-end gap-2 pt-2 border-t border-slate-800">
            <button
              type="button"
              onClick={onClose}
              className="px-4 py-2 bg-slate-800 hover:bg-slate-700 text-slate-300 rounded-xl font-bold"
            >
              ยกเลิก
            </button>
            <button
              type="submit"
              className="px-5 py-2 bg-emerald-600 hover:bg-emerald-500 text-white rounded-xl font-bold shadow-lg shadow-emerald-600/30"
            >
              💾 บันทึกใบลา
            </button>
          </div>
        </form>
      </div>
    </div>
  );
}

// Attendance Record Modal Component
// ----------------------------------------------------
function AttendanceModal({ members, onSave, onClose }) {
  const [employeeName, setEmployeeName] = useState(members[0]?.name || '');
  const [date, setDate] = useState(new Date().toISOString().substring(0, 10));
  const [type, setType] = useState('⏰ มาสาย');
  const [lateMinutes, setLateMinutes] = useState(15);
  const [fineAmount, setFineAmount] = useState(75);
  const [note, setNote] = useState('');

  const handleTypeChange = (newType) => {
    setType(newType);
    if (newType === '🚫 ขาดงาน') {
      setLateMinutes(0);
      setFineAmount(1000);
    } else {
      setLateMinutes(15);
      setFineAmount(75);
    }
  };

  const handleSubmit = (e) => {
    e.preventDefault();
    onSave({
      id: 'att-' + Date.now(),
      employeeName,
      date,
      type,
      lateMinutes: Number(lateMinutes) || 0,
      fineAmount: Number(fineAmount) || 0,
      note
    });
  };

  return (
    <div className="fixed inset-0 z-50 bg-slate-950/80 backdrop-blur-sm flex items-center justify-center p-4">
      <div className="bg-slate-900 border border-slate-800 w-full max-w-md rounded-2xl p-6 space-y-4 shadow-2xl animate-modal text-xs">
        <div className="flex items-center justify-between border-b border-slate-800 pb-3">
          <h3 className="font-extrabold text-white text-base flex items-center gap-2">
            <span>⏰ บันทึก ขาด / มาสาย</span>
          </h3>
          <button onClick={onClose} className="text-slate-400 hover:text-white text-base">✕</button>
        </div>

        <form onSubmit={handleSubmit} className="space-y-4">
          <div className="space-y-1">
            <label className="text-slate-300 font-bold">ชื่อพนักงาน:</label>
            <select
              value={employeeName}
              onChange={(e) => setEmployeeName(e.target.value)}
              className="w-full bg-slate-950 border border-slate-800 rounded-xl p-2.5 text-slate-100 outline-none"
            >
              {members.map(m => (
                <option key={m.id} value={m.name}>{m.name}</option>
              ))}
            </select>
          </div>

          <div className="grid grid-cols-2 gap-3">
            <div className="space-y-1">
              <label className="text-slate-300 font-bold">ประเภท:</label>
              <select
                value={type}
                onChange={(e) => handleTypeChange(e.target.value)}
                className="w-full bg-slate-950 border border-slate-800 rounded-xl p-2.5 text-slate-100 outline-none font-semibold"
              >
                <option value="⏰ มาสาย">⏰ มาสาย</option>
                <option value="🚫 ขาดงาน">🚫 ขาดงาน</option>
              </select>
            </div>
            <div className="space-y-1">
              <label className="text-slate-300 font-bold">วันที่บันทึก:</label>
              <input
                type="date"
                value={date}
                onChange={(e) => setDate(e.target.value)}
                className="w-full bg-slate-950 border border-slate-800 rounded-xl p-2.5 text-slate-100 outline-none font-mono"
              />
            </div>
          </div>

          {type === '⏰ มาสาย' && (
            <div className="grid grid-cols-2 gap-3">
              <div className="space-y-1">
                <label className="text-slate-300 font-bold">นาทีที่สาย:</label>
                <input
                  type="number"
                  value={lateMinutes}
                  onChange={(e) => {
                    const mins = Number(e.target.value) || 0;
                    setLateMinutes(mins);
                    setFineAmount(mins * 5);
                  }}
                  className="w-full bg-slate-950 border border-slate-800 rounded-xl p-2.5 text-slate-100 outline-none font-mono"
                />
              </div>
              <div className="space-y-1">
                <label className="text-slate-300 font-bold">ยอดเงินหักค่าปรับ (บาท):</label>
                <input
                  type="number"
                  value={fineAmount}
                  onChange={(e) => setFineAmount(e.target.value)}
                  className="w-full bg-slate-950 border border-slate-800 rounded-xl p-2.5 text-rose-400 outline-none font-mono font-bold"
                />
              </div>
            </div>
          )}

          {type === '🚫 ขาดงาน' && (
            <div className="space-y-1">
              <label className="text-slate-300 font-bold">ยอดเงินหักค่าปรับขาดงาน (บาท):</label>
              <input
                type="number"
                value={fineAmount}
                onChange={(e) => setFineAmount(e.target.value)}
                className="w-full bg-slate-950 border border-slate-800 rounded-xl p-2.5 text-rose-400 outline-none font-mono font-bold"
              />
            </div>
          )}

          <div className="space-y-1">
            <label className="text-slate-300 font-bold">หมายเหตุ:</label>
            <input
              type="text"
              placeholder="เช่น รถติดหนัก / ไม่แจ้งลา..."
              value={note}
              onChange={(e) => setNote(e.target.value)}
              className="w-full bg-slate-950 border border-slate-800 rounded-xl p-2.5 text-slate-100 outline-none"
            />
          </div>

          <div className="flex items-center justify-end gap-2 pt-2 border-t border-slate-800">
            <button
              type="button"
              onClick={onClose}
              className="px-4 py-2 bg-slate-800 hover:bg-slate-700 text-slate-300 rounded-xl font-bold"
            >
              ยกเลิก
            </button>
            <button
              type="submit"
              className="px-5 py-2 bg-rose-600 hover:bg-rose-500 text-white rounded-xl font-bold shadow-lg shadow-rose-600/30"
            >
              💾 บันทึกสถิติ
            </button>
          </div>
        </form>
      </div>
    </div>
  );
}
